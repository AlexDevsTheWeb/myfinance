<action>
1. **Create new component file** `src/components/TransactionError.tsx`:
   ```typescript
   import { useState, useEffect } from 'react';
   import { Snackbar, Alert, Button, Box } from '@mui/material';
   import { useFinanceStore } from '../store/useFinanceStore';
   
   export function TransactionError() {
     const { saveError, clearSaveError, isSaving } = useFinanceStore();
     const [open, setOpen] = useState(false);
     
     // Show snackbar when error appears
     useEffect(() => {
       if (saveError) {
         setOpen(true);
       }
     }, [saveError]);
     
     const handleClose = (_: any, reason?: string) => {
       if (reason !== 'clickaway') {
         setOpen(false);
         clearSaveError();
       }
     };
     
     return (
       <Snackbar
         open={open}
         autoHideDuration={6000}
         onClose={handleClose}
         anchorOrigin={{ vertical: 'bottom', horizontal: 'center' }}
       >
         <Alert
           onClose={handleClose}
           severity="error"
           variant="filled"
           sx={{ width: '100%' }}
           action={
             <Button
               color="inherit"
               size="small"
               onClick={() => {
                 // Dismiss but don't clear - allows retry
                 setOpen(false);
               }}
             >
               DISMISS
             </Button>
           }
         >
           {saveError}
         </Alert>
       </Snackbar>
     );
   }
   ```
