    import { useEffect, useRef } from 'react';
    import dayjs from 'dayjs';
    import { useAuthStore } from '../store/useAuthStore';
    import { useInvestmentStore } from '../store/useInvestmentStore';

    const PAC_DAY = 1; // Default: 1st of month (configurable per broker)

    /**
     * PAC Automation Initialization Hook (D-04)
     * 
     * On app init, checks each broker account:
     * 1. Does the broker have monthlyPacAmount > 0?
     * 2. Has PAC already been generated this month for this broker?
     * 3. Has the configured PAC day passed this month?
     * 
     * If all conditions met, generates a pending PAC transaction via the store.
     * Uses useRef guard to prevent duplicate generation on re-render/HMR (Pitfall 2).
     */
    export function usePacAutomation() {
      const { user } = useAuthStore();
      const hasChecked = useRef(false);

      useEffect(() => {
        if (!user) {
          hasChecked.current = false;
          return;
        }

        // Prevent duplicate runs (Pitfall 2: HMR guard)
        if (hasChecked.current) return;
        hasChecked.current = true;

        const store = useInvestmentStore.getState();
        const { brokerAccounts, pendingPacTransaction, addPendingPacTransaction } = store;

        // Only run if no pending PAC already exists
        if (pendingPacTransaction) return;

        const today = dayjs();
        const currentMonthKey = today.format('YYYY-MM');

        for (const broker of brokerAccounts) {
          // Skip brokers without PAC configured
          if (!broker.monthlyPacAmount || broker.monthlyPacAmount <= 0) continue;

          const storageKey = `pac_last_${broker.id}`;

          // Check localStorage for per-broker tracking
          const lastPacMonth = localStorage.getItem(storageKey);
          if (lastPacMonth === currentMonthKey) continue;

          // Also check store-level lastPacGenerationDate (backup check)
          if (store.lastPacGenerationDate === currentMonthKey) continue;

          // Default PAC day is 1st of month
          const pacDay = PAC_DAY; // Future: could be per-broker config

          if (today.date() >= pacDay) {
            addPendingPacTransaction({
              brokerId: broker.id,
              amount: broker.monthlyPacAmount,
              date: today.format('YYYY-MM-DD'),
              status: 'pending',
            });

            // Mark this month as generated (persist across refreshes)
            localStorage.setItem(storageKey, currentMonthKey);
            break; // Only generate one pending PAC at a time
          }
        }
      }, [user]);
    }
    ```

    The hook:
    - Reads user from auth store (dependency)
    - Uses useRef guard (hasChecked) to prevent duplicate generation (Pitfall 2 fix)
    - Checks each broker for monthlyPacAmount > 0
    - Uses localStorage for per-broker PAC tracking (persists across sessions)
    - Uses store's lastPacGenerationDate as secondary check
    - Default PAC day = 1st of month
    - Generates only ONE pending PAC at a time (break after first match)
    - Adds pending transaction via addPendingPacTransaction

    const TICKER_REGEX = /^[A-Z0-9]{1,10}(\.[A-Z]{2,3})?$/i;

    export function validateTicker(ticker: string): { valid: boolean; error?: string } {
      if (!ticker?.trim()) {
        return { valid: false, error: 'Ticker is required' };
      }
      if (!TICKER_REGEX.test(ticker.trim())) {
        return {
          valid: false,
          error: 'Invalid ticker format. Expected format: SYMBOL or SYMBOL.EXCHANGE (e.g., SWDA.MI, VWCE.DE, AAPL)',
        };
      }
      return { valid: true };
    }

    export async function validateTickerWithApi(ticker: string): Promise<{ valid: boolean; warning?: string }> {
      const regexResult = validateTicker(ticker);
      if (!regexResult.valid) return regexResult;

      try {
        const { fetchQuote } = await import('../../hooks/useMarketData');
        const quote = await fetchQuote(ticker);
        if (!quote?.regularMarketPrice) {
          return {
            valid: true,
            warning: `Ticker ${ticker} passes format check but could not be verified. Prices may not load.`,
          };
        }
        return { valid: true };
      } catch {
        return { valid: true, warning: 'Could not validate ticker. Check your connection.' };
      }
    }
    ```

    Note: The regex is permissive (D-11 says "lightweight regex pre-check").
    `/^[A-Z0-9]{1,10}(\.[A-Z]{2,3})?$/i` allows 1-10 alphanumeric chars + optional 2-3 letter exchange suffix.
    This covers US stocks (AAPL), European ETFs (SWDA.MI, VWCE.DE), and other common formats.

## Trust Boundaries

| Browser → localStorage | Per-broker PAC last-generation month tracking |

## STRIDE Threat Register

| Threat ID | Category | Component | Severity | Disposition | Mitigation Plan |
|-----------|----------|-----------|----------|-------------|-----------------|
| T-12-11 | Tampering | PAC duplicate generation from HMR or rapid navigation | medium | mitigate | useRef guard (hasChecked.current) + localStorage per-broker month tracking + store lastPacGenerationDate (triple guard) |
| T-12-12 | Tampering | Ticker regex injection (path traversal in symbol) | low | mitigate | Regex `/^[A-Z0-9]{1,10}(\.[A-Z]{2,3})?$/i` allows only alphanumeric and dot — no path traversal, no script injection |
| T-12-13 | Spoofing | localStorage tampering (user edits pac_last_* key) | low | accept | localStorage is client-side only; tampering can only prevent PAC generation (max impact: missed PAC buy for one month). Server-side PAC tracking is deferred per high-level decision. |
| T-12-14 | Information Disclosure | Ticker validation API call leaks user's held tickers to yfin.dev | low | accept | yfin.dev is the existing market data provider (D-14). Ticker symbols are not sensitive financial data (publicly traded ETFs). |
</threat_model>
