2. **Validate backup structure:**
   ```typescript
   function validateBackup(data: unknown): { valid: boolean; error?: string } {
     const backup = data as { version?: string; app?: string; data?: object };
     if (!backup.version) return { valid: false, error: 'Invalid backup: missing version' };
     if (!backup.app || backup.app !== 'myfinance') return { valid: false, error: 'Invalid backup: not a MyFinance backup file' };
     if (!backup.data) return { valid: false, error: 'Invalid backup: missing data' };
     return { valid: true };
   }
   ```

3. **Implementation:**
   ```typescript
   importAllData: async (fileOrData: File | object) => {
     const userId = useAuthStore.getState().user?.uid;
     if (!userId) return;

     set({ saveError: null, isSaving: true });
     try {
       let backup: { data: Partial<FinanceState> };
       
       if (fileOrData instanceof File) {
         const text = await fileOrData.text();
         backup = JSON.parse(text);
       } else {
         backup = fileOrData as { data: Partial<FinanceState> };
       }

       const validation = validateBackup(backup);
       if (!validation.valid) {
         set({ saveError: validation.error, isSaving: false });
         return false;
       }

       const { data } = backup;

       // Batch update all data fields
       const docRef = doc(db, 'users', userId);
       await updateDoc(docRef, {
         initialBalance: data.initialBalance ?? 0,
         accounts: data.accounts ?? [],
         transactions: data.transactions ?? [],
         recurringTransactions: data.recurringTransactions ?? [],
         categories: data.categories ?? [],
         incomeCategories: data.incomeCategories ?? [],
         enabledModules: data.enabledModules ?? { financeTracker: true, carManagement: false, utilityTracker: false },
         balanceStartDate: data.balanceStartDate ?? '2026-01-01',
         carMileage: data.carMileage ?? [],
         carInitialMileage: data.carInitialMileage ?? 0,
         tireSettings: data.tireSettings ?? { summerModel: '', winterModel: '', initialTireType: 'summer' },
         tireChanges: data.tireChanges ?? [],
       });

      // Update local state
      set({
        initialBalance: data.initialBalance ?? 0,
        accounts: data.accounts ?? [],
        transactions: data.transactions ?? [],
        recurringTransactions: data.recurringTransactions ?? [],
        categories: data.categories ?? [],
        incomeCategories: data.incomeCategories ?? [],
        enabledModules: data.enabledModules ?? { financeTracker: true, carManagement: false, utilityTracker: false },
        balanceStartDate: data.balanceStartDate ?? '2026-01-01',
        carMileage: data.carMileage ?? [],
        carInitialMileage: data.carInitialMileage ?? 0,
        tireSettings: data.tireSettings ?? { summerModel: '', winterModel: '', initialTireType: 'summer' },
        tireChanges: data.tireChanges ?? [],
        isSaving: false,
      });

      return true;
     } catch (err) {
       const errorMessage = err instanceof Error ? err.message : 'Failed to import backup';
       set({ saveError: errorMessage, isSaving: false });
       return false;
     }
   }
   ```

| Boundary | Description |
|----------|-------------|
| user → app | Untrusted file input from user |
| app → download | Generated JSON goes to user |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-04-01 | Tampering | importAllData | mitigate | Validate backup structure (version, app name, data object) before restore |
| T-04-02 | Denial of Service | Large file import | mitigate | File size check, consider chunked processing for very large datasets |
| T-04-03 | Information Disclosure | Backup file contains PII | accept | User's own data, no external exposure |
| T-04-04 | Elevation of Privilege | Restore overwrites other user data | mitigate | Restore only modifies authenticated user's own document |
</threat_model>
