```typescript
}
```

2. Add to initial state:
```typescript

```

3. Add the setLanguage action:
```typescript
setLanguage: (lang: string) => {
  // Save to localStorage for persistence
  localStorage.setItem('myfinance_language', lang);
  // Change i18n language
  i18n.changeLanguage(lang);
  set({ language: lang });
},
```

```typescript

```
